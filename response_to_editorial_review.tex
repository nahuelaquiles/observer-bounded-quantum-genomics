\subsection*{1. Theorem 2 Commuting Projector Hypothesis, Instantaneous Cancellation, and Readout Labels}
\textbf{Audit Finding:} State the commuting energy-diagonal hypothesis $[\Pi_{\mathrm{mut}}, H_0] = 0$ explicitly in Theorem~2 (fixing the cancellation-free DC static calculation), clarify instantaneous vs.\ cycle-averaged first-order behavior, and align the projector label on $1.60 \times 10^{-28}$ between the Main Text and Supplementary Text~S1.2.

\medskip\noindent\textbf{Resolution:} Fully resolved in Theorem~2, Supplementary Text~S1.2, \texttt{scripts/01}, Fig.~1C, and Table~1:
\begin{itemize}[leftmargin=1.6em,itemsep=2pt]
    \item \textbf{Exact Instantaneous First-Order Vanishing for $[\Pi_{\mathrm{mut}}, H_0] = 0$:} Theorem~2 now explicitly states $0 \le \Pi_{\mathrm{mut}} \le I$ and $[\Pi_{\mathrm{mut}}, H_0] = 0$ ($\Pi_{\varepsilon_1(0)} = |\varepsilon_1(0)\rangle\langle\varepsilon_1(0)|$). Because $[\delta H(t), \rho_\beta]$ has strictly zero diagonal entries in the $H_0$ eigenbasis and the KMS Davies--Lindblad generator $\Lcal_0$ decouples eigenbasis populations from off-diagonal coherences, $\rho^{(1)}(t)$ is purely off-diagonal in the $H_0$ eigenbasis at \emph{every instant $t$}, so $\Tr(\Pi_{\varepsilon_1(0)}\rho^{(1)}(t)) = 0$ identically without requiring cycle averaging (verified numerically across $64$ drive phases: $\max_t |\Tr(\Pi_{\varepsilon_1(0)}\rho^{(1)}(t))| = 1.26 \times 10^{-15}$).
    \item \textbf{Cancellation-Free DC Shift \& Harmonized Projector Table:} Using \texttt{expm1} in \texttt{scripts/01} eliminates float cancellation at $E_0 \le 10^{-6}\,\mathrm{V\,m^{-1}}$, giving an exact DC shift of $3.73 \times 10^{-15}$ vs.\ asymptotic Kubo $2\lambda/(k_B T) = 7.48 \times 10^{-15}$ at $1\,\mu\mathrm{V\,m^{-1}}$. All texts and tables now report the exact projector for each number at $1\,\mu\mathrm{V\,m^{-1}}$: fixed excited-eigenstate $\Pi_{\varepsilon_1(0)}$ gives $1.26 \times 10^{-26}$ (Floquet Drazin, crossover $4.48 \times 10^5\,\mathrm{V\,m^{-1}}$) and $1.60 \times 10^{-28}$ (adiabatic KMS, crossover $3.98 \times 10^6\,\mathrm{V\,m^{-1}}$); instantaneous $|\varepsilon_1(t)\rangle\langle\varepsilon_1(t)|$ gives $3.48 \times 10^{-30}$ (adiabatic KMS, crossover $2.70 \times 10^7\,\mathrm{V\,m^{-1}}$); and non-commuting bare chemical tautomer $|\mathrm{G}^*\text{:}\mathrm{C}^*\rangle\langle\mathrm{G}^*\text{:}\mathrm{C}^*|$ gives cycle-averaged $2.33 \times 10^{-30}$ (Floquet, crossover $3.30 \times 10^7\,\mathrm{V\,m^{-1}}$) and $8.51 \times 10^{-32}$ (KMS, crossover $1.72 \times 10^8\,\mathrm{V\,m^{-1}}$).
\end{itemize}
