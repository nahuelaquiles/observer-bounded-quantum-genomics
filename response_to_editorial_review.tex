\section*{Point-by-Point Resolution of Review Findings}

\subsection*{1. Human Pancreatic Islet \texttt{GSE218316} Mapping, Timepoint Consistency, and Disjoint Pooling}
\textbf{Audit Finding:} In the previous version of \texttt{scripts/08}, mapping gene names from an HVG table across raw Cell Ranger indices omitted detected clock/BER genes, pooled multiple exposure durations, and drew ensembles with replacement inside folds; when aligned per library on donor-matched 24-hour samples with strictly disjoint pools, all $22$ target genes are detected, Clock-10 alone fails cross-donor LODO generalization ($\mathrm{AUC} = 0.347$ at $\nu = 100$), while the combined 22-gene Clock + BER/NRF2 panel achieves strong cross-donor recovery ($\mathrm{CV\ AUC} = 1.000$, $\mathrm{LODO\ AUC} = 0.960$ at $\nu = 100$).

\medskip\noindent\textbf{Resolution:} Fully adopted and verified from raw GEO tarballs (\texttt{GSM6740278}, \texttt{GSM6740279}, \texttt{GSM6740282}). Each donor-matched 24-hour 10x Cell Ranger matrix ($n = 2{,}465$ cells: $1{,}187$ untreated, $1{,}278$ $\mathrm{H_2O_2}$ across $3$ donors) is aligned directly to its own \texttt{features.tsv.gz} gene table, recovering all $10$ core clock transcripts and all $12$ BER/NRF2 redox-repair transcripts across all three libraries (\texttt{data/module6\_sc\_h2o2\_gene\_erasure.csv}). At $\nu = 1$, mean per-gene clock zero-UMI dropout is $85.71\%$ ($73.27\%$ \emph{CLOCK} to $92.05\%$ \emph{ARNTL}; $30.34\%$ of cells carry zero UMIs across all $10$ clock genes simultaneously; $\text{Clock CV AUC} = 0.519$ / $\text{LODO } 0.486$; $\text{Combined CV AUC} = 0.583$ / $\text{LODO } 0.529$). Partitioning each fold into strictly disjoint cell pools up to $\nu = 100$ (\texttt{data/module6\_sc\_h2o2\_ensemble\_scaling.csv} and Fig.~6B) shows transparently that unsynchronized single-timepoint Clock-10 pools reach within-cohort $\text{CV AUC} = 0.750$ ($\nu = 50$) and $0.650$ ($\nu = 100$) but fail cross-donor $\mathrm{LODO}$ ($\text{AUC} = 0.398$ and $0.347$), whereas combining Clock-10 with the $12$ BER/NRF2 redox-repair genes achieves $\text{CV AUC} = 0.860$ / $\text{LODO AUC} = 0.937$ ($\nu = 50$) and $\text{CV AUC} = 1.000$ / $\text{LODO AUC} = 0.960$ ($\nu = 100$). Row~3 in Table~4 is explicitly labeled \textbf{Exploratory ($n=3$ donors, 24-h $\Pi_{\mathcal{O}}$)}, and \texttt{GDS3342} has been removed.

\subsection*{2. Theorem 1 Hermiticity Preservation and Theorem 2 Scope \& KMS Projector Conventions}
\textbf{Audit Finding:} (a) Fig.~1B previously illustrated Theorem~1 using a random fast-sector matrix that did not preserve Hermiticity; (b) Theorem~2 was phrased as a universal Lindblad bound whereas its static $2\lambda/(k_B T)$ factor is asymptotic ($\lambda \ll k_B T$) and its KMS curvature depends on whether $\Pi_{\mathrm{mut}}$ is held fixed in the bare basis or rotated with instantaneous eigenstates.

\medskip\noindent\textbf{Resolution:}
\begin{itemize}[leftmargin=1.6em,itemsep=2pt]
    \item \textbf{Exact Bare-Basis GKSL Perturbation on $\Pcal$:} \texttt{scripts/01} and Fig.~1B now construct $\delta\Lcal[\rho] = \sum_{a=0}^2 \delta\gamma_a (P_a \rho P_a - \frac{1}{2}\{P_a, \rho\})$ with bare-state projectors $P_a = |a\rangle\langle a|$, which is a bona fide trace- and Hermiticity-preserving GKSL generator ($\|\delta\Lcal[\rho^\dagger] - (\delta\Lcal[\rho])^\dagger\|_F = 0.0$, trace error $0.0$) supported strictly on $\Pcal$ ($\delta\Lcal_{QQ} = \delta\Lcal_{QP} = \delta\Lcal_{PQ} = 0$), yielding $\|\Delta M_{0,1,2}\|_2 = 0.0$ and $\|\Delta M_3\|_2 = 4.667 \times 10^{39}\,\mathrm{s}^{-3}$ (matching $\|\Lcal_{QP}(\delta\Lcal_{PP})\Lcal_{PQ}\|_2$ to relative error $7.77 \times 10^{-16}$).
    \item \textbf{Scoped Finite-State KMS Theorem 2 \& Dual Adiabatic Readout:} Theorem~2 and Supplementary Text~S1.2 are now explicitly scoped to finite non-degenerate KMS-compliant Davies--Lindblad generators (and evaluated numerically across all $E_0 \in [10^{-9}, 10^7]\,\mathrm{V\,m^{-1}}$ on the 3-state G:C model), state the asymptotic Kubo static scaling $|\Delta\mu_{\mathrm{DC}}/\mu_0| \le (2\lambda/k_B T)(1 + \mathcal{O}(\lambda/k_B T))$ alongside the exact matrix DC shift ($3.63 \times 10^{-15}$ at $1\,\mu\mathrm{V\,m^{-1}}$), and plot and tabulate \emph{both} adiabatic KMS conventions alongside the fixed-bath Floquet Drazin shift ($1.26 \times 10^{-26}$, $50\,\mathrm{mK}$ crossover $4.48 \times 10^5\,\mathrm{V\,m^{-1}}$): fixed bare-tautomer projector $|1\rangle\langle 1|$ ($1.60 \times 10^{-28}$, crossover $3.98 \times 10^6\,\mathrm{V\,m^{-1}}$) and instantaneous eigenstate projector $|\varepsilon_1(\epsilon)\rangle\langle\varepsilon_1(\epsilon)|$ ($3.48 \times 10^{-30}$, crossover $2.70 \times 10^7\,\mathrm{V\,m^{-1}}$). Every regime lies $\ge 23$ orders of magnitude below the $50\,\mathrm{mK}$ thermal floor ($2.53 \times 10^{-3}$) at $1\,\mu\mathrm{V\,m^{-1}}$.
\end{itemize}

\subsection*{3. Single-Cell Binary Occupancy Entropy vs.\ Gaussian Fisher Proxy $I_{\mathrm{GF}}(\nu)$ (Module 2)}
\textbf{Audit Finding:} Distinguish Proposition~1's snapshot binary lesion-occupancy entropy $h_2(1 - e^{-\lambda_{\mathrm{cell}}})$ from the Clarke--Barron asymptotic mutual information formula, which is an approximation at $\nu = 1$, and align sensing-to-memory crossover indexing.

\medskip\noindent\textbf{Resolution:} Proposition~1 now explicitly reports binary snapshot occupancy entropy $h_2(1 - e^{-\lambda_{\mathrm{cell}}}) = 5.09 \times 10^{-3}$, $1.34 \times 10^{-2}$, and $3.76 \times 10^{-2}\,\mathrm{bits}$ for $\bar{p}_{\oxog} \in \{10^{-6}, 3\times 10^{-6}, 10^{-5}\}$ ($P(N_{\mathrm{lesion}}=0) = 0.9996, 0.9988, 0.9960$), while Theorem~4 defines $I_{\mathrm{GF}}(\nu) \coloneqq \frac{1}{2}\log_2\det(I_K + \nu\sigma_\theta^2\overline{F}_1)$ explicitly as the \textbf{Gaussian Fisher information proxy} ($I_{\mathrm{GF}}(1) = 4.68 \times 10^{-3}\,\mathrm{bits}$ at $\nu=1$, converging to Clarke--Barron mutual information up to $\mathcal{O}(1)$ at large $\nu$). Memory-limited onset ($D_{\mathrm{mem}} > D_{\mathrm{sens}}$) is uniformly reported at $K = 1, 1, 2, 5$ harmonic modes ($0, 0, 1, 4$ sensing-limited modes) for $C_B \in \{2, 4, 8, 16\}\,\mathrm{bits}$.

\subsection*{4. Primary Adult Liver (\texttt{UBERON:0002107}) AlphaGenome Scoring \& Human $\oxog$ Risk Scope}
\textbf{Audit Finding:} \texttt{scripts/04} previously allowed \texttt{EFO:0001187} HepG2 fallback tracks when summarizing liver effects, and the $13$-variant clock-promoter panel should be framed as exploratory alongside a clear statement that prospective $\Delta R^2$ benchmarking against human $\oxog$ and mutation maps remains open.

\medskip\noindent\textbf{Resolution:} Removed \texttt{EFO:0001187} from \texttt{scripts/04} and re-ran live AlphaGenome scoring strictly on \texttt{UBERON:0002107} (adult human liver) across all $13$ variants and all $6$ modalities, reporting both nominal Mann--Whitney $p$-values and $6$-modality Benjamini--Hochberg $q$-values (\texttt{DNASE} $12.74\times$, $p = 0.0451$, $q = 0.186$; \texttt{ATAC} $7.67\times$, $p = 0.0932$, $q = 0.186$; \texttt{RNA\_SEQ} $1.85\times$, $p = 0.0653$, $q = 0.186$; Table~S2 and Fig.~4D). Both Results and Limitations explicitly frame Module~4 as exploratory \emph{in silico} concordance complementing living-cell saturation MPRA (\texttt{GSE126550}, $n = 541$ SNVs) and identify incremental $\Delta R^2$ prediction of human $\oxog$ lesion maps and somatic/germline $\mathrm{G{:}C \to T{:}A}$ mutations over sequence-context and motif baselines as a key future test.

\subsection*{5. Standalone Reproducibility, Nucleosome Step Coupling, and Exploratory Cosmic/Beamline Framing}
\textbf{Audit Finding:} Archive portable human promoter sequences so \texttt{scripts/03} runs without external genome caches, avoid treating adjacent base-pair steps of \texttt{1KX5} as independent replicates in a Mann--Whitney test, separate diploid-DNA vs.\ whole-cell solar-neutrino event counts, and clearly separate exploratory astrodynamic/beamline modeling (Module~5 + Row~5b) from empirically tested terrestrial stages.

\medskip\noindent\textbf{Resolution:}
\begin{itemize}[leftmargin=1.6em,itemsep=2pt]
    \item \textbf{Portable Promoter Shards \& Centralized Path Resolver:} Archived all $4{,}570$ \texttt{GRCh38} MANE Select $[-500,+500]\,\mathrm{bp}$ promoter sequences directly inside the repository (\texttt{data/human\_grch38\_mane\_promoters\_part1.csv.gz} and \texttt{part2.csv.gz}, $<680\,\mathrm{KB}$ each) and wired \texttt{scripts/analysis\_paths.py} + \texttt{OBQG\_RAW\_DIR} across all scripts.
    \item \textbf{Descriptive Nucleosome Sterics:} Removed the step-wise Mann--Whitney $p$-value on \texttt{1KX5}, reporting descriptive clash means ($54.24$ outward vs.\ $785.32$ inward octamer-core clashes, $35.1\%$ outward zero-clash poses vs.\ $100\%$ inward clashing, $8.29\times$ modeled accessibility ratio) alongside the $0$-clash native cryo-EM structures (\texttt{8VWT}, \texttt{8VWV}, \texttt{9EOZ}).
    \item \textbf{Unambiguous Neutrino Targets \& Exploratory Module 5 / Row 5b Labels:} Reported diploid-DNA ($4.15 \times 10^{-15}\,\mathrm{yr^{-1}}$, $1.08 \times 10^{19}$ diploid genomes for $5\sigma$) and whole-cell ($6.86 \times 10^{-13}\,\mathrm{yr^{-1}}$, $6.53 \times 10^{16}$ cells / $65\,\mathrm{t}$ biomass for $5\sigma$) solar $\nu_e$ scattering side by side, labeled $\sigma_{\mathrm{ion}}/\sigma_{\nu e} = 2.71 \times 10^{27}$ strictly as an electromagnetic-to-weak cross-section ratio, and marked Module~5 and Row~5b explicitly as \textbf{Exploratory Modeling / Exploratory Beamline Falsification Specification} across all headings, captions, and tables.
\end{itemize}
