\section*{Supplementary Text S1: Operator-Theoretic Proofs, KMS Detailed-Balance Generator, and 3D Cylindrical B-DNA Helical Multipole Equations}

\subsection*{S1.1 Exact Kubo--Martin--Schwinger (KMS) Stationarity and Lumer--Phillips Coercivity of $\Lcal_0$}
Let $H_0 = \sum_{\alpha=0}^{2} \varepsilon_\alpha |\varepsilon_\alpha\rangle\langle\varepsilon_\alpha|$ denote the solvated three-state G:C protonic Hamiltonian in its exact orthonormal eigenbasis ($\varepsilon_0 < \varepsilon_1 < \varepsilon_2$) (\textit{1}, \textit{2}, \textit{3}). Constructing jump operators $L_{\alpha\beta} = \sqrt{\gamma_{\alpha\beta}}\,|\varepsilon_\alpha\rangle\langle\varepsilon_\beta|$ ($\alpha \neq \beta$) satisfying $\gamma_{\alpha\beta} / \gamma_{\beta\alpha} = \exp[-\beta(\varepsilon_\alpha - \varepsilon_\beta)]$ together with eigenbasis pure-dephasing operators $D_\alpha = \sqrt{\gamma_{\mathrm{deph}}}\,|\varepsilon_\alpha\rangle\langle\varepsilon_\alpha|$, the GKSL generator $\Lcal_0[\rho] = -\frac{i}{\hbar}[H_0, \rho] + \sum_{\alpha\neq\beta}\bigl(L_{\alpha\beta}\rho L_{\alpha\beta}^\dagger - \frac{1}{2}\{L_{\alpha\beta}^\dagger L_{\alpha\beta}, \rho\}\bigr) + \sum_\alpha\bigl(D_\alpha\rho D_\alpha^\dagger - \frac{1}{2}\{D_\alpha^\dagger D_\alpha, \rho\}\bigr)$ decouples eigenbasis populations $p_\alpha = \langle\varepsilon_\alpha|\rho|\varepsilon_\alpha\rangle$ from off-diagonal eigenbasis coherences $\rho_{\alpha\beta}$. Evaluating $\Lcal_0[\rho_\beta]$ on $\rho_\beta = Z^{-1}\sum_\alpha e^{-\beta\varepsilon_\alpha}|\varepsilon_\alpha\rangle\langle\varepsilon_\alpha|$ yields:
\begin{equation}
    \langle\varepsilon_\alpha|\Lcal_0[\rho_\beta]|\varepsilon_\alpha\rangle = Z^{-1}\sum_{\beta\neq\alpha}\Bigl(\gamma_{\alpha\beta}e^{-\beta\varepsilon_\beta} - \gamma_{\beta\alpha}e^{-\beta\varepsilon_\alpha}\Bigr) = 0,
\end{equation}
while every fast-subspace coherence decays with spectral gap $g(\Lcal_{PP}) = 1.0110 \times 10^{13}\,\mathrm{s}^{-1}$ ($\tau_{\mathrm{mem}} = 98.91\,\mathrm{fs}$; numerical Frobenius stationarity error $\|\rho_{\mathrm{ss}} - \rho_\beta\|_F = 1.61 \times 10^{-11}$). Moreover, the symmetric part of the fast-subspace block $\Lcal_{PP}$ satisfies strict Lumer--Phillips numerical coercivity $g_2(\Lcal_{PP}) \coloneqq -\lambda_{\max}((\Lcal_{PP}+\Lcal_{PP}^\dagger)/2) = 1.0092 \times 10^{13}\,\mathrm{s}^{-1} > 0$ ($\kappa_2(V) = 1.000093$), guaranteeing $\|e^{t\Lcal_{PP}}\|_2 \le e^{-g_2(\Lcal_{PP})t}$ and $\|(s\mathcal{I}-\Lcal_{PP})^{-1}\|_2 \le (\mathrm{Re}\,s + g_2(\Lcal_{PP}))^{-1}$ with unit prefactor ($C = 1$). By contrast, phenomenological site-basis dephasing does not commute with $H_0$ when $J_{\alpha\beta} \neq 0$, violating detailed balance and inflating apparent excited-state tautomer populations toward the infinite-temperature limit.

\subsection*{S1.2 Fixed-Bath Floquet--Lindblad Drazin vs.\ Adiabatic KMS Rectification Bounds (Theorem 2)}
For $\Lcal(t) = \Lcal_0 + \cos(\omega t)\Lcal_V$ with $\Lcal_V[\rho] = -\frac{i}{\hbar}[V_{\mathrm{dip}}, \rho]$ and $\|\Lcal_V\| \sim \lambda/\hbar \ll g(\Lcal_0)$, expanding the periodic Floquet steady state $\rho_{\mathrm{per}}(t) = \rho_\beta + \rho^{(1)}(t) + \rho^{(2)}(t) + \cdots$ under a \emph{fixed-bath dissipator} $\Lcal_0$ gives $\rho^{(1)}(t) = \frac{1}{2}\bigl[(i\omega\mathcal{I} - \Lcal_0)^{-1}\Lcal_V[\rho_\beta]e^{i\omega t} + (-i\omega\mathcal{I} - \Lcal_0)^{-1}\Lcal_V[\rho_\beta]e^{-i\omega t}\bigr]$, whose time average over macromolecular replication windows $T_{\mathrm{rep}} \gg 2\pi/\omega$ vanishes identically ($\overline{\rho^{(1)}} = 0$). The time-averaged second-order state satisfies $\Lcal_0[\overline{\rho^{(2)}}] = -\overline{\cos(\omega t)\Lcal_V[\rho^{(1)}(t)]}$, yielding
\begin{equation}
    \overline{\rho^{(2)}_{\mathrm{Floquet}}} = -\frac{1}{2}\Lcal_0^+\Lcal_V\,\mathrm{Re}\!\Bigl[(i\omega\mathcal{I} - \Lcal_0)^{-1}\Bigr]\Lcal_V[\rho_\beta],
\end{equation}
where $\Lcal_0^+$ is the Drazin pseudoinverse on the trace-zero subspace. Evaluating Eq.~(S2) directly on the $9 \times 9$ Liouvillian yields $|\overline{\Delta\mu_{\mathrm{Floquet}}}/\mu_0| = 1.263 \times 10^{-26}$ at $E_0 = 1\,\mu\mathrm{V\,m^{-1}}$ ($\omega = 10^9\,\mathrm{rad\,s^{-1}}$; $1.264 \times 10^{-26}$ at $\omega = 10^{13}\,\mathrm{rad\,s^{-1}}$; scaling as $\mathcal{O}(\lambda^2 / [\hbar^2\gamma_{\mathrm{pop}}\gamma_{\mathrm{deph}}])$, with $50\,\mathrm{mK}$ crossover $E_0 \approx 4.48 \times 10^5\,\mathrm{V\,m^{-1}}$). Conversely, in the \emph{adiabatic instantaneous-bath KMS limit} ($\omega \ll \gamma_{\mathrm{bath}}$, where jump rates track the instantaneous dressed Hamiltonian $H(t) = H_0 + \epsilon\cos(\omega t)V_{\mathrm{dip}}$), time averaging yields $\overline{\rho^{(2)}_{\mathrm{KMS}}} = \frac{1}{4}\left.\frac{\mathrm{d}^2\rho_\beta(\epsilon)}{\mathrm{d}\epsilon^2}\right|_{\epsilon=0} = \mathcal{O}((\lambda / k_B T)^2)$, giving $|\overline{\Delta\mu_{\mathrm{KMS}}}/\mu_0| = 3.484 \times 10^{-30}$ at $E_0 = 1\,\mu\mathrm{V\,m^{-1}}$ ($50\,\mathrm{mK}$ crossover $E_0 \approx 2.70 \times 10^7\,\mathrm{V\,m^{-1}}$) (\textit{4}).

\subsection*{S1.3 Complete 3D Cylindrical Helical B-DNA Electrostatic Field and Mineral-Terrace Geometry}
Modeling double-stranded B-DNA (radius $R_{\mathrm{DNA}} = 1.0\,\mathrm{nm}$, helical pitch $P = 3.4\,\mathrm{nm}$, wavenumber $k = 2\pi/P = 1.848\,\mathrm{nm}^{-1}$, double-helix chirality index $m = +2$ for right-handed D-DNA and $m = -2$ for left-handed L-DNA) via screened cylindrical potential $\Phi(r,\phi,z) = R_0(r) + R_1(r)\cos(kz - m\phi)$ (\textit{5}, \textit{23}), all three cylindrical field components $\mathbf{E} = -\nabla\Phi$ (together with the relativistic Thomas precession factor $1/2$ in spin--orbit coupling) are retained simultaneously in the $240{,}000$-trajectory integrator:
\begin{equation}
    E_r(r,\phi,z) = -R_0'(r) - R_1'(r)\cos(kz - m\phi), \quad
    E_\phi(r,\phi,z) = -\frac{m}{r}R_1(r)\sin(kz - m\phi), \quad
    E_z(r,\phi,z) = k R_1(r)\sin(kz - m\phi).
\end{equation}
Along a relativistic lepton trajectory of polar angle $\psi$ to the helix axis ($\dot{z} = v\cos\psi$, $r\dot{\phi} = v\sin\psi\sin\alpha$), the helical phase advances at $\dot{\Theta} = kv\cos\psi - (m/r)v\sin\psi\sin\alpha$, changing sign across the critical pitch-resonance cone $\tan\psi_{\mathrm{crit}} = kr / |m| = 0.508$ ($\psi_{\mathrm{crit}} = 26.9^\circ$). For B-DNA duplexes adsorbed \emph{flat (horizontally)} on mica or clay mineral terraces via divalent-cation phosphate bridging, a zenith-collimated $\cos^2\theta_z$ cosmic-ray shower satisfies $\cos\psi = \sin\theta_z\cos\phi_h$, placing $98.29\%$ of incident rays broadside at $\psi > \psi_{\mathrm{crit}}$ ($1.281\times$ inward deflection gain over isotropic solution). Under parity transformation $(m, \mathcal{L}) \mapsto (-m, -\mathcal{L})$, radial deflection onto the central purine stack is strictly invariant, whereas enantiomer inversion $m \mapsto -m$ at fixed cosmic-ray helicity $\mathcal{L}$ reverses the chiral perturbation identically ($\langle\Delta r\rangle_{m=+2} + \langle\Delta r\rangle_{m=-2} = 0.0\,\mathrm{m}$).

\section*{Supplementary Text S2: Two-Tier Accelerator Protocol for Open Falsification Row 5b}

    \item \textbf{Enantiomeric and Repair-Uncollapsed Targets:} (a) Ultra-high-vacuum self-assembled monolayers (SAMs, $0.5\,\mathrm{cm^2}$ on template-stripped $\mathrm{Au(111)}$) of $5'$-thiolated synthetic right-handed D-DNA versus mirror-image L-DNA $21$-bp hairpins carrying a central \texttt{5'-GGGG-3'} HOMO hole sink, mounted on a precision rotary goniometer tilted at $\psi_1 = 15^\circ$ ($< \psi_{\mathrm{crit}}$) versus $\psi_2 = 45^\circ$ ($> \psi_{\mathrm{crit}}$); and (b) hydrated thin monolayers of GO/MMR-triple-deficient \emph{E.\ coli} $\Delta\textit{mutM}\Delta\textit{mutY}\Delta\textit{mutS}$ (uncollapsing $\mathrm{G{:}C \to T{:}A}$ to $98.62\%$; Table~S3).
    \item \textbf{Two-Tier Readout and Exact Poisson Event Budget:} Quantify site-specific $\oxog$ excision yield via Fpg cleavage / LC-MS/MS on D- versus L-hairpins and $\mathrm{G{:}C \to T{:}A}$ transversion asymmetry via UMI Duplex-Seq (error floor $< 10^{-7}\,\mathrm{bp^{-1}}$), using $N_{\min}(|\mathcal{A}|) = z_\alpha^2 / |\mathcal{A}|^2$ per helicity arm:
    \begin{itemize}
        \item \emph{Tier 1 --- Resonant CISS / Low-Energy DEA Exchange ($|\mathcal{A}_{\mathrm{eff}}| \ge 10^{-3}$):} $3\sigma$ ($5\sigma$) detection requires $N_{\min} = 9.0 \times 10^6$ ($2.5 \times 10^7$) single-lesion events per helicity arm.     \end{itemize}
\end{enumerate}
